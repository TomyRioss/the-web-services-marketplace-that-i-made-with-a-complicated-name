# Job Hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Daily-updated hub that aggregates job listings from Computrabajo, ZonaJobs, and Reddit r/empleos into one filterable UI.

**Architecture:** Next.js App Router site. `app/api/scrape` orchestrates three isolated scraper modules (Selenium for Computrabajo/ZonaJobs, Reddit JSON search API), upserts normalized results into a Supabase `jobs` table (dedup on `url`). A GitHub Actions cron job calls the endpoint daily. `app/page.tsx` reads from Supabase and renders a shadcn-based grid.

**Tech Stack:** Next.js 16 / React 19 / TypeScript, Tailwind v4, shadcn/ui, `@supabase/supabase-js`, `selenium-webdriver` + `chromedriver`, Vitest for unit tests.

## Global Constraints

- Nunca CSS puro — TailwindCSS only, never touch `global.css` (project rule).
- Usar shadcn para componentes prefabricados generales (project rule).
- Paleta de color: blanco/naranja (project rule).
- Siempre responsive mobile + PC (project rule).
- Ningún componente/archivo mayor a 500 líneas — modularizar (project rule).
- Todo error catcheado con feedback de consola Y visual (project rule).
- DB changes ya autorizadas explícitamente por el usuario para este proyecto — no pedir permiso de nuevo dentro de este plan.

```typescript
import type { JobResult } from "./types";

interface RedditSearchResponse {
  data: {
    children: Array<{ data: { title: string; permalink: string } }>;
  };
}

export async function scrapeReddit(keywords: string[]): Promise<JobResult[]> {
  const results: JobResult[] = [];

  for (const keyword of keywords) {
    const url = `https://www.reddit.com/r/empleos/search.json?q=${encodeURIComponent(
      keyword
    )}&restrict_sr=1&sort=new&limit=25`;

    try {
      const res = await fetch(url, {
        headers: { "User-Agent": "ttmcobuild-job-hub/1.0" },
      });
      if (!res.ok) continue;

      const json = (await res.json()) as RedditSearchResponse;
      for (const child of json.data.children) {
        results.push({
          source: "reddit",
          title: child.data.title,
          company: null,
          location: null,
          url: `https://www.reddit.com${child.data.permalink}`,
          keywordMatched: keyword,
        });
      }
    } catch {
      continue;
    }
  }

  return results;
}
```

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";
import type { JobResult } from "../scrapers/types";

export async function upsertJobs(
  client: SupabaseClient,
  jobs: JobResult[]
): Promise<{ count: number }> {
  if (jobs.length === 0) {
    return { count: 0 };
  }

  const rows = jobs.map((job) => ({
    source: job.source,
    title: job.title,
    company: job.company,
    location: job.location,
    url: job.url,
    keyword_matched: job.keywordMatched,
  }));

  const { error } = await client.from("jobs").upsert(rows, { onConflict: "url" });

  if (error) {
    throw new Error(`Supabase upsert failed: ${error.message}`);
  }

  return { count: rows.length };
}
```

```typescript
import { Builder, By, type WebDriver } from "selenium-webdriver";
import chrome from "selenium-webdriver/chrome";
import type { JobResult } from "./types";

interface WebDriverLike {
  get(url: string): Promise<void>;
  findElements(locator: { css: string }): Promise<
    Array<{ findElement(locator: { css: string }): Promise<{ getText(): Promise<string>; getAttribute(name: string): Promise<string> }> }>
  >;
  quit(): Promise<void>;
}

async function defaultDriverFactory(): Promise<WebDriverLike> {
  const options = new chrome.Options();
  options.addArguments("--headless=new", "--no-sandbox", "--disable-dev-shm-usage");
  const driver: WebDriver = await new Builder().forBrowser("chrome").setChromeOptions(options).build();
  return driver as unknown as WebDriverLike;
}

export async function scrapeComputrabajo(
  keywords: string[],
  location: string,
  driverFactory: () => Promise<WebDriverLike> = defaultDriverFactory
): Promise<JobResult[]> {
  const results: JobResult[] = [];

  for (const keyword of keywords) {
    const driver = await driverFactory();
    try {
      const searchUrl = `https://ar.computrabajo.com/trabajo-de-${encodeURIComponent(
        keyword.replace(/\s+/g, "-")
      )}-en-${encodeURIComponent(location.replace(/\s+/g, "-"))}`;
      await driver.get(searchUrl);

      const cards = await driver.findElements({ css: "article.box_offer" });
      for (const card of cards) {
        const linkEl = await card.findElement({ css: "a.js-o-link" });
        const url = await linkEl.getAttribute("href");
        const title = await linkEl.getText();
        const companyEl = await card.findElement({ css: ".fs13" });
        const company = await companyEl.getText();

        results.push({
          source: "computrabajo",
          title,
          company: company || null,
          location: null,
          url,
          keywordMatched: keyword,
        });
      }
    } catch {
      // isolated per keyword — continue with the next one
    } finally {
      await driver.quit();
    }
  }

  return results;
}
```

```typescript
import { NextResponse } from "next/server";
import { scrapeReddit } from "../../../lib/scrapers/reddit";
import { scrapeComputrabajo } from "../../../lib/scrapers/computrabajo";
import { scrapeZonajobs } from "../../../lib/scrapers/zonajobs";
import { getSupabaseServerClient } from "../../../lib/supabase/client";
import { upsertJobs } from "../../../lib/supabase/jobs";
import { KEYWORDS, LOCATION } from "../../../lib/config/search";
import type { JobResult } from "../../../lib/scrapers/types";

interface SourceResult {
  ok: boolean;
  count: number;
  error?: string;
}

async function runSource(
  name: string,
  fn: () => Promise<JobResult[]>,
  client: ReturnType<typeof getSupabaseServerClient>
): Promise<SourceResult> {
  try {
    const jobs = await fn();
    const { count } = await upsertJobs(client, jobs);
    return { ok: true, count };
  } catch (err) {
    console.error(`[scrape:${name}] failed`, err);
    return { ok: false, count: 0, error: err instanceof Error ? err.message : String(err) };
  }
}

export async function POST(req: Request): Promise<Response> {
  const secret = req.headers.get("x-scrape-secret");
  if (!secret || secret !== process.env.SCRAPE_SECRET) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  const client = getSupabaseServerClient();

  const [computrabajo, zonajobs, reddit] = await Promise.all([
    runSource("computrabajo", () => scrapeComputrabajo(KEYWORDS, LOCATION), client),
    runSource("zonajobs", () => scrapeZonajobs(KEYWORDS, LOCATION), client),
    runSource("reddit", () => scrapeReddit(KEYWORDS), client),
  ]);

  return NextResponse.json({ computrabajo, zonajobs, reddit });
}
```

```typescript
import { getSupabaseServerClient } from "@/lib/supabase/client";
import { JobCard } from "@/components/job-card";

export const revalidate = 0;

interface JobRow {
  id: string;
  source: string;
  title: string;
  company: string | null;
  location: string | null;
  url: string;
  scraped_at: string;
}

export default async function Home() {
  const client = getSupabaseServerClient();
  const { data, error } = await client
    .from("jobs")
    .select("id, source, title, company, location, url, scraped_at")
    .order("scraped_at", { ascending: false })
    .limit(200);

  const jobs = (data ?? []) as JobRow[];
  const lastUpdated = jobs[0]?.scraped_at ?? null;

  return (
    <main className="min-h-screen bg-white px-4 py-8 sm:px-8">
      <div className="mx-auto max-w-5xl">
        <h1 className="text-2xl font-bold text-orange-600 sm:text-3xl">Ofertas de trabajo</h1>
        <p className="mt-1 text-sm text-muted-foreground">
          {lastUpdated
            ? `Última actualización: ${new Date(lastUpdated).toLocaleString("es-AR")}`
            : "Todavía no hay ofertas cargadas."}
        </p>

        {error && (
          <div className="mt-4 rounded-md border border-orange-400 bg-orange-50 px-4 py-3 text-sm text-orange-800">
            No se pudieron cargar las ofertas. Reintentá en unos minutos.
          </div>
        )}

        <div className="mt-6 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
          {jobs.map((job) => (
            <JobCard
              key={job.id}
              job={{
                title: job.title,
                company: job.company,
                location: job.location,
                url: job.url,
                source: job.source,
                scrapedAt: job.scraped_at,
              }}
            />
          ))}
        </div>
      </div>
    </main>
  );
}
```
